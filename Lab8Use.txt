using System;
using System.Collections.Generic;
using RealtorAgency;

namespace Lab8Use
{
    class Program
    {
        static AgencyManager manager = new AgencyManager();

        static void Main(string[] args)
        {
            Console.OutputEncoding = System.Text.Encoding.UTF8;
            Console.InputEncoding = System.Text.Encoding.UTF8;

            manager.LoadTestData();

            while (true)
            {
                Console.Clear();
                Console.WriteLine("1. Управління клієнтами");
                Console.WriteLine("2. Управління нерухомістю");
                Console.WriteLine("3. Управління пропозиціями");
                Console.WriteLine("4. Пошук");
                Console.WriteLine("0. Вихід");
                Console.Write("Виберіть пункт: ");

                string choice = Console.ReadLine();

                try
                {
                    switch (choice)
                    {
                        case "1":
                            ClientsMenu();
                            break;
                        case "2":
                            PropertiesMenu();
                            break;
                        case "3":
                            OffersMenu();
                            break;
                        case "4":
                            SearchMenu();
                            break;
                        case "0":
                            return;
                        default:
                            Console.WriteLine("Невірний ввід.");
                            Console.ReadKey();
                            break;
                    }
                }
                catch (FormatException)
                {
                    Console.WriteLine("Помилка: невірний формат вводу числа.");
                    Console.ReadKey();
                }
                catch (ArgumentException ex)
                {
                    Console.WriteLine($"Помилка: {ex.Message}");
                    Console.ReadKey();
                }
                catch (InvalidOperationException ex)
                {
                    Console.WriteLine($"Помилка: {ex.Message}");
                    Console.ReadKey();
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"Помилка: {ex.Message}");
                    Console.ReadKey();
                }
            }
        }

        static void ClientsMenu()
        {
            Console.Clear();
            Console.WriteLine("Клієнти");
            Console.WriteLine("1. Показати всіх");
            Console.WriteLine("2. Показати конкретного клієнта");
            Console.WriteLine("3. Додати клієнта");
            Console.WriteLine("4. Видалити клієнта");
            Console.WriteLine("5. Змінити дані клієнта");
            Console.WriteLine("6. Сортувати за ім'ям");
            Console.WriteLine("7. Сортувати за прізвищем");
            Console.WriteLine("8. Сортувати за першою цифрою рахунку");
            Console.WriteLine("0. Назад");

            switch (Console.ReadLine())
            {
                case "1":
                    PrintList(manager.Clients);
                    break;
                case "2":
                    Console.Write("Введіть Id клієнта: ");
                    int idShow = int.Parse(Console.ReadLine());
                    Console.WriteLine(manager.GetClientById(idShow).ToString());
                    Console.ReadKey();
                    break;
                case "3":
                    Client c = new Client();
                    Console.Write("Ім'я: ");
                    c.FirstName = Console.ReadLine();
                    Console.Write("Прізвище: ");
                    c.LastName = Console.ReadLine();
                    Console.Write("Рахунок: ");
                    c.BankAccountNumber = Console.ReadLine();
                    Console.WriteLine("Типи: 1-Однушка, 2-Двушка, 3-Трьошка, 4-Ділянка");
                    Console.Write("Бажаний тип: ");
                    c.DesiredPropertyType = (PropertyType)int.Parse(Console.ReadLine());
                    Console.Write("Максимальна ціна: ");
                    c.MaxDesiredPrice = decimal.Parse(Console.ReadLine());

                    manager.AddClient(c);
                    Console.WriteLine("Додано.");
                    Console.ReadKey();
                    break;
                case "4":
                    Console.Write("Введіть Id клієнта для видалення: ");
                    int idDel = int.Parse(Console.ReadLine());
                    manager.RemoveClientById(idDel);
                    Console.WriteLine("Видалено.");
                    Console.ReadKey();
                    break;
                case "5":
                    Console.Write("Введіть Id клієнта для зміни: ");
                    int idUpd = int.Parse(Console.ReadLine());

                    Console.Write("Нове ім'я: ");
                    string newFirst = Console.ReadLine();
                    Console.Write("Нове прізвище: ");
                    string newLast = Console.ReadLine();
                    Console.Write("Новий рахунок: ");
                    string newAcc = Console.ReadLine();

                    manager.UpdateClientById(idUpd, newFirst, newLast, newAcc);
                    Console.WriteLine("Оновлено.");
                    Console.ReadKey();
                    break;
                case "6":
                    PrintList(manager.GetClientsSortedByName());
                    break;
                case "7":
                    PrintList(manager.GetClientsSortedByLastName());
                    break;
                case "8":
                    PrintList(manager.GetClientsSortedByBankAccount());
                    break;
                case "0":
                    return;
            }
        }

        static void OffersMenu()
        {
            Console.Clear();
            Console.WriteLine("Пропозиції");
            Console.WriteLine("1. Додати об'єкт до списку пропозицій клієнта");
            Console.WriteLine("2. Перевірити доступність бажаного об'єкта для клієнта");
            Console.WriteLine("3. Відхилити пропозицію");
            Console.WriteLine("0. Назад");

            switch (Console.ReadLine())
            {
                case "1":
                    Console.Write("Введіть Id клієнта: ");
                    int cId = int.Parse(Console.ReadLine());
                    Console.Write("Введіть Id об'єкта нерухомості: ");
                    int pId = int.Parse(Console.ReadLine());

                    manager.AddOfferToClient(cId, pId);
                    Console.WriteLine("Запропоновано.");
                    Console.ReadKey();
                    break;
                case "2":
                    Console.Write("Введіть Id клієнта: ");
                    int clId = int.Parse(Console.ReadLine());

                    bool isAvail = manager.CheckIfDesiredAvailable(clId);
                    Console.WriteLine($"Для клієнта підходящі об'єкти: " + (isAvail ? "Є" : "НЕМАЄ"));
                    Console.ReadKey();
                    break;
                case "3":
                    Console.Write("Введіть Id клієнта: ");
                    int cRejId = int.Parse(Console.ReadLine());

                    Client clientInfo = manager.GetClientById(cRejId);
                    if (clientInfo.OfferedProperties.Count == 0)
                    {
                        Console.WriteLine("У клієнта немає пропозицій.");
                        Console.ReadKey();
                        return;
                    }

                    Console.WriteLine("Поточні пропозиції клієнта:");
                    foreach (var prop in clientInfo.OfferedProperties)
                    {
                        Console.WriteLine($"Id: {prop.Id} | Тип: {prop.Type} | Ціна: {prop.Price}");
                    }

                    Console.Write("Введіть Id пропозиції для відхилення: ");
                    int oId = int.Parse(Console.ReadLine());

                    manager.RemoveOfferFromClient(cRejId, oId);
                    Console.WriteLine("Відхилено.");
                    Console.ReadKey();
                    break;
                case "0":
                    return;
            }
        }

        static void SearchMenu()
        {
            Console.Clear();
            Console.WriteLine("Пошук");
            Console.WriteLine("1. По клієнтах (за ключовим словом)");
            Console.WriteLine("2. По нерухомості (за ключовим словом в описі)");
            Console.WriteLine("3. Глобальний пошук (клієнти + нерухомість)");
            Console.WriteLine("4. Розширений (прізвище + тип)");
            Console.WriteLine("0. Назад");

            Console.Write("Вибір: ");
            string choice = Console.ReadLine();

            if (choice == "0")
            {
                return;
            }

            switch (choice)
            {
                case "1":
                    Console.Write("Введіть слово для пошуку: ");
                    string word1 = Console.ReadLine();
                    PrintList(manager.SearchClients(word1));
                    break;
                case "2":
                    Console.Write("Введіть слово для пошуку: ");
                    string word2 = Console.ReadLine();
                    PrintList(manager.SearchProperties(word2));
                    break;
                case "3":
                    Console.Write("Введіть слово для глобального пошуку: ");
                    string word3 = Console.ReadLine();

                    Console.WriteLine("Клієнти:");
                    var clientResults = manager.SearchClients(word3);
                    if (clientResults.Count == 0)
                    {
                        Console.WriteLine("Пусто.");
                    }
                    foreach (var item in clientResults)
                    {
                        Console.WriteLine(item.ToString());
                    }

                    Console.WriteLine("Нерухомість:");
                    var propGlobalResults = manager.SearchProperties(word3);
                    if (propGlobalResults.Count == 0)
                    {
                        Console.WriteLine("Пусто.");
                    }
                    foreach (var item in propGlobalResults)
                    {
                        Console.WriteLine(item.ToString());
                    }

                    Console.WriteLine("Натисніть будь-яку клавішу...");
                    Console.ReadKey();
                    break;
                case "4":
                    Console.WriteLine("Типи: 1-Однушка, 2-Двушка, 3-Трьошка, 4-Ділянка");
                    Console.Write("Необхідний тип (цифра): ");
                    PropertyType type = (PropertyType)int.Parse(Console.ReadLine());
                    Console.Write("Прізвище: ");
                    string lastName = Console.ReadLine();
                    PrintList(manager.AdvancedSearch(lastName, type));
                    break;
            }
        }

        static void PrintList<T>(List<T> list)
        {
            Console.WriteLine("Результат:");
            if (list.Count == 0)
            {
                Console.WriteLine("Пусто.");
            }
            foreach (var item in list)
            {
                Console.WriteLine(item.ToString());
            }
            Console.WriteLine("Натисніть будь-яку клавішу...");
            Console.ReadKey();
        }
    }
}
